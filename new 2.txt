prompt = """
You are generating synthetic test data for AML/SAR analytics.

Create a dataset with the following requirements:

- Exactly 10 records
- Output strictly as CSV (no explanations, no markdown)
- Include a header row

Columns:
Case_ID,Filing_Date,SAR_Category,SAR_SubCategory,Narrative_Text

Requirements:
- Case_ID: CASE0001-CASE0010
- Filing_Date: realistic (YYYY-MM-DD)
- SAR_Category: Fraud, Money Laundering, Structuring, Terrorist Financing
- SAR_SubCategory: realistic subtype
- Narrative_Text:
    - 100-200 words
    - formal SAR tone
    - no real names (use "Subject A", "Customer", etc.)
    - include suspicious behavior patterns
    - DO NOT include commas that break CSV format

Return only CSV.
"""

# ------------------------------
# CALL MODEL
# ------------------------------
response = client.chat.completions.create(
    model="gpt-5.2-2025-12-II-dztpm",   # your deployed model name
    messages=[{"role": "user", "content": prompt}],
    temperature=0.3
)

csv_output = response.choices[0].message.content

# ------------------------------
# SAVE FILE (Databricks)
# ------------------------------
file_path = "/dbfs/FileStore/sar_mock_data.csv"

with open(file_path, "w") as f:
    f.write(csv_output)

print("File saved to:", file_path)
print(csv_output[:500])   # preview
